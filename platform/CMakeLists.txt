if(WIN32)
  set(RECOMP_OS_SOURCE os_win32.cpp)
else()
  set(RECOMP_OS_SOURCE os_posix.cpp)
endif()
add_library(recomp_platform OBJECT ${RECOMP_OS_SOURCE} env.cpp profile_markers.cpp)
target_include_directories(recomp_platform PUBLIC ${RECOMP_ROOT})
if(WIN32)
  # os_win32.cpp resolves host names through Winsock's getaddrinfo.
  target_link_libraries(recomp_platform PUBLIC ws2_32)
endif()
target_compile_options(recomp_platform PRIVATE ${RECOMP_WARN_STRICT})
recomp_optimize(recomp_platform 2)
if(ANDROID)
  # Bionic provides pthread functions in libc, without a separate libpthread.
  target_link_libraries(recomp_platform PUBLIC dl log)
elseif(NOT WIN32 AND NOT APPLE)
  target_link_libraries(recomp_platform PUBLIC dl pthread)
endif()

if(NOT IOS AND NOT ANDROID)
add_executable(platform_tests tests/platform_tests.cpp)
target_link_libraries(platform_tests PRIVATE recomp_platform)
target_compile_options(platform_tests PRIVATE ${RECOMP_WARN_STRICT})
recomp_optimize(platform_tests 1)
recomp_test_binary(platform_tests)
add_test(NAME platform_tests COMMAND platform_tests WORKING_DIRECTORY ${RECOMP_ROOT})
set_tests_properties(platform_tests PROPERTIES LABELS nogame)
endif()
