cmake_minimum_required(VERSION 3.20)
project(ir_ssa_checks C)
include("${CMAKE_CURRENT_LIST_DIR}/../../harness_platform.cmake")
foreach(check_target ir_ssa_checks ir_ssa_null_checks)
  add_executable(${check_target} "${CMAKE_BINARY_DIR}/generated.c" support.c ../../corpus/fragments/harness.c ${KIT_OS_SOURCES})
  target_include_directories(${check_target} PRIVATE "${KIT_RUNTIME}" "${KIT_ROOT}" "${CMAKE_BINARY_DIR}")
  target_compile_definitions(${check_target} PRIVATE FIXTURE_CHECK_ONLY=1)
  target_compile_options(${check_target} PRIVATE -O2 -ffp-contract=off -fno-builtin -Wall -Wextra -Wno-unused)
  set_target_properties(${check_target} PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
  target_link_libraries(${check_target} PRIVATE ${KIT_OS_LIBRARIES})
endforeach()
target_compile_definitions(ir_ssa_null_checks PRIVATE RECOMP_NULL_CHECKS=1)
