cmake_minimum_required(VERSION 3.20)
project(x87_locals_experiment C)
include("${CMAKE_CURRENT_LIST_DIR}/../../harness_platform.cmake")
add_executable(x87_locals "${CMAKE_BINARY_DIR}/generated.c" harness.c ${KIT_OS_SOURCES})
target_include_directories(x87_locals PRIVATE "${KIT_RUNTIME}" "${KIT_ROOT}" "${CMAKE_BINARY_DIR}")
target_compile_options(x87_locals PRIVATE -O2 -ffp-contract=off -Wall -Wextra)
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
  target_compile_options(x87_locals PRIVATE -save-temps=obj)
endif()
target_link_libraries(x87_locals PRIVATE ${KIT_OS_LIBRARIES})
