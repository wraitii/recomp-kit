cmake_minimum_required(VERSION 3.20)
project(x87_locals_experiment C)
add_executable(x87_locals "${CMAKE_BINARY_DIR}/generated.c" harness.c)
target_include_directories(x87_locals PRIVATE "${KIT_RUNTIME}" "${CMAKE_BINARY_DIR}")
target_compile_options(x87_locals PRIVATE -O2 -ffp-contract=off -Wall -Wextra)
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
  target_compile_options(x87_locals PRIVATE -save-temps=obj)
endif()
if(NOT WIN32)
  target_link_libraries(x87_locals PRIVATE m)
endif()
