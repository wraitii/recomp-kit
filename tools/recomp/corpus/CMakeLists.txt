cmake_minimum_required(VERSION 3.20)
project(function_corpus C)
include("${CMAKE_CURRENT_LIST_DIR}/../harness_platform.cmake")
include("${CMAKE_BINARY_DIR}/sources.cmake")
add_executable(function_corpus ${CORPUS_SOURCES} harness.c boundaries.c ${KIT_OS_SOURCES})
target_include_directories(function_corpus PRIVATE "${KIT_RUNTIME}" "${KIT_ROOT}" "${CMAKE_BINARY_DIR}" "${CORPUS_FIXTURES}")
target_compile_options(function_corpus PRIVATE -O2 -ffp-contract=off -fno-builtin -Wall -Wextra -Wno-unused)
set_target_properties(function_corpus PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
option(CORPUS_ASAN "Build the corpus with AddressSanitizer (correctness runs only)" OFF)
if(CORPUS_ASAN)
  target_compile_options(function_corpus PRIVATE -g -fsanitize=address)
  target_link_options(function_corpus PRIVATE -fsanitize=address)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
target_link_libraries(function_corpus PRIVATE ${KIT_OS_LIBRARIES})
if(DEFINED CORPUS_STORE_HOOKS)
  target_compile_definitions(function_corpus PRIVATE RECOMP_STORE_HOOKS=${CORPUS_STORE_HOOKS})
endif()
