cmake_minimum_required(VERSION 3.20)
project(function_corpus C)
include("${CMAKE_BINARY_DIR}/sources.cmake")
add_executable(function_corpus ${CORPUS_SOURCES} harness.c boundaries.c)
target_include_directories(function_corpus PRIVATE "${KIT_RUNTIME}" "${CMAKE_BINARY_DIR}" "${CORPUS_FIXTURES}")
target_compile_options(function_corpus PRIVATE -O2 -ffp-contract=off -fno-builtin -Wall -Wextra -Wno-unused)
set_target_properties(function_corpus PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT WIN32)
  target_link_libraries(function_corpus PRIVATE m)
endif()
