cmake_minimum_required(VERSION 3.20)
project(cpu_locals_checks C)
include("${CMAKE_CURRENT_LIST_DIR}/../../harness_platform.cmake")
foreach(target cpu_locals cpu_locals_null_checks)
  add_executable(${target} "${CMAKE_BINARY_DIR}/generated.c" support.c ../../corpus/fragments/harness.c ${KIT_OS_SOURCES})
  target_include_directories(${target} PRIVATE "${KIT_RUNTIME}" "${KIT_ROOT}" "${CMAKE_BINARY_DIR}")
  target_compile_definitions(${target} PRIVATE FIXTURE_CHECK_ONLY=1)
  target_compile_options(${target} PRIVATE -O2 -ffp-contract=off -Wall -Wextra)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_options(${target} PRIVATE -save-temps=obj)
  endif()
  target_link_libraries(${target} PRIVATE ${KIT_OS_LIBRARIES})
endforeach()
target_compile_definitions(cpu_locals_null_checks PRIVATE RECOMP_NULL_CHECKS=1)
